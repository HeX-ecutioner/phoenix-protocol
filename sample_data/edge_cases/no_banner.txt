! Phoenix Protocol Sample: No Security Banner
! Valid Cisco configuration with hostname, but omitting MOTD and login banners
version 15.6
hostname NO-BANNER-RTR-01
!
service timestamps debug datetime msec
service timestamps log datetime msec
service password-encryption
!
no ip http server
no ip http secure-server
no service tcp-small-servers
no service udp-small-servers
no ip source-route
!
! Deliberately no banner motd or banner login statements
!
username secadmin privilege 15 secret 9 $9$dummyNoBannerScryptHash12345
enable secret 9 $9$dummyNoBannerScryptEnableHash12345
!
ip domain-name lab.internal
ip ssh version 2
login block-for 300 attempts 3 within 60
logging host 198.51.100.50
ntp server 198.51.100.123
!
ip access-list standard MGMT_ACL
 permit 192.0.2.0 0.0.0.255
 deny any log
!
line vty 0 4
 access-class MGMT_ACL in
 transport input ssh
!
line vty 5 15
 access-class MGMT_ACL in
 transport input ssh
!
end
