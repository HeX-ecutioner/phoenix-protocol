! Phoenix Protocol Sample: Insecure Services Explicitly Disabled
! Tests rule NET-008 pass when HTTP server and small servers are explicitly disabled
version 15.7
hostname INSEC-SERVICES-OFF-RTR
!
service timestamps debug datetime msec
service timestamps log datetime msec
service password-encryption
!
! Insecure / legacy services explicitly turned OFF
no ip http server
no ip http secure-server
no service tcp-small-servers
no service udp-small-servers
no ip source-route
!
banner motd ^C
AUTHORIZED USERS ONLY
^C
!
username secadmin privilege 15 secret 9 $9$dummySvcOffScryptHash12345
enable secret 9 $9$dummySvcOffScryptEnableHash12345
!
ip domain-name lab.internal
ip ssh version 2
login block-for 300 attempts 3 within 60
logging host 198.51.100.50
ntp server 198.51.100.123
!
ip access-list standard MGMT_ACL
 permit 192.0.2.0 0.0.0.255
 deny any log
!
line vty 0 4
 access-class MGMT_ACL in
 transport input ssh
!
end
