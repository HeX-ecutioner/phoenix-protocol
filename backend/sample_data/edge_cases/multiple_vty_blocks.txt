! Phoenix Protocol Sample: Multiple VTY Blocks (All Secure)
! Tests scoped multi-range VTY block parsing with distinct configurations
version 15.7
hostname SEC-MULTIVTY-RTR
!
service timestamps debug datetime msec
service timestamps log datetime msec
service password-encryption
!
no ip http server
no ip http secure-server
no service tcp-small-servers
no service udp-small-servers
no ip source-route
!
banner motd ^C
===================================================
SEC-MULTIVTY-RTR - ALL ACCESS MONITORED
===================================================
^C
!
username secadmin privilege 15 secret 9 $9$dummyMultiVtyScryptHash12345
enable secret 9 $9$dummyMultiVtyScryptEnableHash12345
!
ip domain-name enterprise.internal
ip ssh version 2
login block-for 300 attempts 3 within 60
logging host 198.51.100.50
ntp server 198.51.100.123
!
ip access-list standard PRIMARY_ADMIN_ACL
 permit 192.0.2.0 0.0.0.255
 deny any log
!
ip access-list standard SECONDARY_ADMIN_ACL
 permit 198.51.100.0 0.0.0.255
 deny any log
!
! First VTY Block (0-4): primary admin access with 5-minute timeout
line vty 0 4
 access-class PRIMARY_ADMIN_ACL in
 exec-timeout 5 0
 transport input ssh
!
! Second VTY Block (5-15): secondary admin access with 10-minute timeout
line vty 5 15
 access-class SECONDARY_ADMIN_ACL in
 exec-timeout 10 0
 transport input ssh
!
end
